Implement handleDragEnd:
```typescript
async function handleDragEnd(event: DragEndEvent) {
  const { active, over } = event
  if (!over || active.id === over.id) return

  const oldIndex = activeTasks.findIndex(t => t.id === active.id)
  const newIndex = activeTasks.findIndex(t => t.id === over.id)
  const reordered = arrayMove(activeTasks, oldIndex, newIndex)

  await db.transaction('rw', db.tasks, async () => {
    for (let i = 0; i < reordered.length; i++) {
      await db.tasks.update(reordered[i].id, { sortOrder: i })
    }
  })
}
```

Per RESEARCH.md Pitfall 4: sortOrder MUST be persisted to Dexie in a transaction, and queries MUST sort by sortOrder. Verify the useLiveQuery query in TaskList uses `.sortBy('sortOrder')`.

exportData function:

```typescript
export async function exportData(): Promise<void> {
  const blob = await exportDB(db, { prettyJson: true })
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = `donotnervous-backup-${new Date().toISOString().slice(0, 10)}.json`
  document.body.appendChild(a)
  a.click()
  document.body.removeChild(a)
  URL.revokeObjectURL(url)
}
```

importData function:
```typescript
export async function importData(file: File): Promise<void> {
  const meta = await peakImportFile(file)
  if (!meta?.data?.databaseName) {
    throw new Error('Invalid backup file')
  }
  await importInto(db, file, {
    acceptNameDiff: true,
    acceptVersionDiff: true,
    clearTablesBeforeImport: true,
    overwriteValues: true,
  })
}
```

These follow the exact pattern from RESEARCH.md lines 313-357.

**4. Create src/hooks/useToast.ts (simple local-state toast hook).**
Create a lightweight toast hook using React local state (NOT a Zustand store). This keeps Phase 1 simple -- toasts are only needed for export/import feedback in the settings drawer.

```typescript
import { useState, useCallback } from 'react'

interface ToastState {
  message: string
  type: 'success' | 'error'
  visible: boolean
}

export function useToast() {
  const [toast, setToast] = useState<ToastState>({
    message: '',
    type: 'success',
    visible: false,
  })

  const showToast = useCallback((message: string, type: 'success' | 'error' = 'success') => {
    setToast({ message, type, visible: true })
    setTimeout(() => {
      setToast((prev) => ({ ...prev, visible: false }))
    }, 3000)
  }, [])

  const hideToast = useCallback(() => {
    setToast((prev) => ({ ...prev, visible: false }))
  }, [])

  return { toast, showToast, hideToast }
}
```

This hook manages: message, type (success/error), visible state, and auto-dismiss after 3 seconds. Used by SettingsDrawer and App.tsx. No external state library needed -- pure React useState + useCallback.
